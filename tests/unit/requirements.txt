# Unit test dependencies, installed by ansible-test units and by CI.
#
# Floors, not pins: ansible-test resolves these inside its own per-python venv
# across the whole units matrix (py3.10 through py3.13), and exact pins would
# make that matrix unsatisfiable. The exact-pin discipline applies to
# requirements-dev.txt, where linter output must be byte-reproducible.
requests>=2.25.0
pytest>=7.0.0
pytest-mock>=3.10.0
# TLS transport tests generate an ephemeral self-signed leaf in-process. This
# keeps private keys out of the repository and avoids depending on an external
# openssl executable in every supported Python matrix cell.
cryptography>=3.4.0
# module_utils/ipmi.py (asmb8_power/asmb8_boot/asmb8_info) wraps this directly
# rather than shelling out to community.general.ipmi_power/ipmi_boot -- see
# that module's docstring. tests/unit/plugins/module_utils/test_ipmi.py mocks
# pyghmi's Command class rather than ever opening a real IPMI session, but it
# still needs the real pyghmi.exceptions types to construct realistic
# failures, so the dependency is real here too, not just at runtime.
pyghmi>=1.5.0
# tomllib entered the standard library in Python 3.11. tests/unit/roles/
# test_autoinstall_answer_template.py parses rendered TOML to prove
# roles/asmb8_autoinstall_iso's answer.toml.j2 output is valid, and needs a
# tomllib-compatible reader on Python 3.10 (still in this collection's units
# matrix) where the stdlib module does not exist yet. This is a stdlib
# backport for that one test file, not a new runtime dependency -- the
# collection itself never imports TOML at runtime. The marker keeps it out of
# the 3.11+ venvs entirely, where the stdlib module already covers it.
tomli>=2.0.0; python_version < "3.11"
